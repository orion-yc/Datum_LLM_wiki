cmake_minimum_required(VERSION 3.21)
project(DatumKnowledge VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Network Sql)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(LibArchive REQUIRED)
find_package(Threads REQUIRED)
add_executable(datum-server backend/src/main.cpp backend/src/core.cpp backend/src/learning.cpp backend/src/team.cpp backend/src/research.cpp backend/src/wiki.cpp backend/src/pdf.cpp backend/src/pdf_figures.cpp backend/src/source_management.cpp backend/src/uploads.cpp)
target_include_directories(datum-server PRIVATE backend/src backend/third_party)
target_sources(datum-server PRIVATE backend/src/projects.cpp)
target_link_libraries(datum-server PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql OpenSSL::Crypto Threads::Threads LibArchive::LibArchive)
if(WIN32)
  target_compile_definitions(datum-server PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX)
  target_link_libraries(datum-server PRIVATE ws2_32 crypt32)
endif()
if(MSVC)
  target_compile_options(datum-server PRIVATE /utf-8 /W4)
else()
  target_compile_options(datum-server PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME core COMMAND datum-server --self-test)
install(TARGETS datum-server RUNTIME DESTINATION bin)
